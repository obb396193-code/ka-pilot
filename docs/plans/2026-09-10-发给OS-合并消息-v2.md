   ```bash
   git fetch origin deploy/web-standalone
   git worktree add ../ka-web-standalone origin/deploy/web-standalone
   cd ../ka-web-standalone && cat SOURCE_SHA      # 应为 12b7fd17（或更新）
   HOSTNAME=0.0.0.0 PORT=3000 node apps/web/server.js   # ENV 与原 web 相同
   ```
